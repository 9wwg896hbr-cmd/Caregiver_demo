function esc(s) { return String(s).replace(/[&<>"']/g, c => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c])) }
function fmt(n) { return Number(n).toLocaleString(lang === "ru" ? "ru-RU" : lang === "uz" ? "uz-UZ" : "en-US") }

function notify(msg) {
  const el = $("#toast");
  el.textContent = msg;
  el.classList.add("show");
  clearTimeout(toastTimer);
  toastTimer = setTimeout(() => el.classList.remove("show"), 3000);
}

function goto(page) {
  $$(".page").forEach(x => x.classList.toggle("active", x.id === page));
  $$(".nav button").forEach(x => x.classList.toggle("active", x.dataset.page === page));
  window.scrollTo({ top: 0, behavior: "smooth" });
}

function aiAnswer(q) {
  let s = q.toLowerCase();
  if (/stroke|инсульт|insult|speech|речь|nutq/.test(s)) return t("aiStroke");
  if (/pressure|давлен|bosim/.test(s)) return t("aiPressure");
  if (/medicine|medication|лекар|таблет|dori/.test(s)) return t("aiMedicine");
  if (/dementia|деменц|demens/.test(s)) return t("aiDementia");
  if (/bed|bedbound|пролеж|лежач|yotoq/.test(s)) return t("aiBed");
  if (/water|hydration|вода|пить|suv|suyuqlik/.test(s)) return t("aiWater");
  return t("aiGeneric");
}

// Global Click Handlers
document.addEventListener("click", e => {
  let b = e.target.closest("[data-page]");
  if (b) goto(b.dataset.page);

  let g = e.target.closest("[data-go]");
  if (g) goto(g.dataset.go);

  let c = e.target.closest("[data-cat]");
  if (c) chooseCategory(c.dataset.cat);

  let p = e.target.closest("[data-profile]");
  if (p) openProfile(p.dataset.profile);

  let a = e.target.closest("[data-article]");
  if (a) {
    let art = articleData().find(x => x.id === Number(a.dataset.article));
    if (art) {
      $("#modalTitle").textContent = art.title;
      $("#modalContent").innerHTML = `
        <p class="text-sm text-[#24372e] leading-relaxed mb-4">${esc(art.body)}</p>
        <div class="bg-[#f6f0df] text-[#75602f] p-3 rounded-xl text-xs mb-4">${t("aiDisclaimer")}</div>
        <button class="btn border border-[#cfdcd0] text-[#526459] hover:bg-[#e7f0e8] text-xs font-semibold px-4 py-2 rounded-xl transition-all" id="articleClose">${t("close")}</button>`;
      $("#modalback").classList.add("open");
      $("#articleClose").onclick = () => $("#modalback").classList.remove("open");
    }
  }

  let d = e.target.closest("[data-delete-entry]");
  if (d) {
    localStorage.setItem("cgEntries", JSON.stringify(getEntries().filter(x => String(x.id) !== d.dataset.deleteEntry)));
    renderJournal();
    notify(t("deleted"));
  }
});

  let q = $("#question").value.trim();
  if (!q) return;
  chatHistory.push({ role: "user", text: q }, { role: "ai", text: aiAnswer(q) });
  $("#question").value = "";
  renderChat();
});

$("#clearChat").onclick = () => { chatHistory = []; renderChat(); };
